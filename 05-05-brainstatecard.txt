            key={stage.id}

                        text-xs font-mono transition-all duration-300
                        ${i === activeStage
                          ? "border"
                          : i < activeStage
                            ? "opacity-60"
                            : "opacity-25"}`}
            style={
              i === activeStage
                ? {
                    background:  `${stage.color}20`,
                    borderColor: `${stage.color}50`,
                    color:        stage.color,
                    boxShadow:   `0 0 15px ${stage.color}30`,
                  }
                : i < activeStage
                  ? { color: `${stage.color}80` }
                  : { color: "#64748b" }
            }
            animate={i === activeStage ? { scale: [1, 1.05, 1] } : {}}
            transition={{ duration: 0.8, repeat: Infinity }}
          >
            {stage.icon}
            <span>{stage.label}</span>

            {/* System badge */}
            {i === activeStage && (
              <span
                className="text-[8px] px-1 py-0.5 rounded"
                style={{ background: `${stage.color}25` }}
              >
                {stage.system}
              </span>
            )}

            {/* Connector arrow */}
            {i < PIPELINE_STAGES.length - 1 && (
              <ArrowRight
                size={10}
                className="ml-0.5"
                style={{ color: i < activeStage ? stage.color : "#374151" }}
              />
            )}
          </motion.div>
        ))}
      </div>
    </div>
  );
